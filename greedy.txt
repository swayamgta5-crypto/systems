def prim(graph, start):

    visited = [start]

    mst = []

    while len(visited) < len(graph):

        min_edge = None

        for u in visited:

            for v, weight in graph[u]:

                if v not in visited:

                    if min_edge is None or weight < min_edge[2]:

                        min_edge = (u, v, weight)

        mst.append(min_edge)

        visited.append(min_edge[1])

    print("Minimum Spanning Tree:")

    for u, v, weight in mst:

        print(u, "-", v, ":", weight)

graph = {

    'A': [('B', 2), ('C', 3)],

    'B': [('A', 2), ('C', 1), ('D', 4)],

    'C': [('A', 3), ('B', 1), ('D', 5)],

    'D': [('B', 4), ('C', 5)]

}

prim(graph, 'A')

edges = [
    (2, 3, 4),
    (0, 3, 5),
    (0, 2, 6),
    (0, 1, 10),
    (1, 3, 15)
]

visited = []
total_cost = 0

print("Edges in Minimum Spanning Tree:")

for u, v, weight in edges:
    if u not in visited or v not in visited:
        print(u, "-", v, ":", weight)
        total_cost += weight
        visited.append(u)
        visited.append(v)

print("Total Cost =", total_cost)
